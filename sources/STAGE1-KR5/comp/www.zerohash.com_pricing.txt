    });
    m.addEventListener('mouseout', function (e) {
      if (!desktop()) return;
      var from = e.target.closest && e.target.closest('.nav_item_dropdown.is-desktop');
      if (!from || !m.contains(from)) return;
      var next = e.relatedTarget;
      if (next && from.contains(next)) return;
      if (next && m.contains(next)) {
        e.stopPropagation();
        var dest = next.closest && next.closest('.nav_item_dropdown.is-desktop');
        if (dest) openOnly(dest);
        return;
      }
      closeAll();
    }, true);
    m.addEventListener('focusin', function (e) {
      if (!desktop()) return;
      var dd = e.target.closest && e.target.closest('.nav_item_dropdown.is-desktop');
      if (dd) openOnly(dd);
    });
    m.addEventListener('focusout', function (e) {
      if (!desktop()) return;
      var next = e.relatedTarget;
      if (next && m.contains(next)) return;
      closeAll();
    });
    m.addEventListener('click', function (e) {
      if (!desktop()) return;
      var toggle = e.target.closest && e.target.closest('.w-dropdown-toggle');
      if (!toggle || !m.contains(toggle)) return;
      var dd = toggle.parentElement;
      setTimeout(function () {
        var list = dd.querySelector(':scope > .w-dropdown-list');
        if (!list) return;
        if (list.classList.contains('w--open')) openOnly(dd);
        else setOpen(dd, false);
      }, 0);
    });
    document.addEventListener('keydown', function (e) {
      if (!desktop() || e.key !== 'Escape') return;
      var open = dropdowns().filter(function (dd) { return dd.classList.contains('w--open'); })[0];
      if (!open) return;
      var toggle = open.querySelector(':scope > .w-dropdown-toggle');
      closeAll();
      if (toggle && open.contains(document.activeElement)) toggle.focus();
    });
  }
  if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', bind);
  else bind();
  window.addEventListener('load', measure);
})();
